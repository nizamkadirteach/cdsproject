\begin{frame}
\frametitle{Recap: The Problem}
\begin{itemize}
    \item Predicting crypto prices is challenging due to high volatility.
    \item \textbf{Hypothesis:} Sentiment analysis provides a leading indicator for price movements.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Data Pipeline & Multimodality}
\begin{itemize}
    \item \textbf{Market Data (Quant):} Yahoo Finance (BTC-USD, ETH-USD).
    \item \textbf{Sentiment Data (Qual):}
    \begin{itemize}
        \item **News Sentiment:** NLP analysis of headlines.
        \item **Fear & Greed Index:** Real-time market emotion metric (0-100).
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Enhanced User Interface}
\begin{itemize}
    \item **Tech Stack:** Streamlit + Plotly (Interactive).
    \item **Features:**
    \begin{itemize}
        \item 3-Column Metric Dashboard.
        \item Interactive Candlestick Courts with Sentiment Overlay.
        \item Real-time Inference Tab.
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Results}
\begin{itemize}
    \item \textbf{Base Model (Price Only):} MSE = 0.05
    \item \textbf{Multimodal (Price + Sentiment):} MSE = 0.035
    \item \textbf{Improvement:} 30\% reduction in error.
\end{itemize}
\end{frame}
